\section{XIAO ESP32-S3}

\begin{frame}{Implementaci\'on sobre Seeed XIAO ESP32-S3}
    \begin{columns}[T]
        \column{0.52\textwidth}
        \footnotesize
        \textbf{Motivaci\'on}
        \begin{itemize}\itemsep0.12cm
            \item Mayor margen de c\'omputo para ML local e inferencia TinyML.
            \item Mejor margen de memoria frente a LVGL, NimBLE, buffers DMA y tareas FreeRTOS.
            \item Plataforma m\'as cercana al prototipo final, con USB-C integrado.
            \item Permite ordenar el dise\~no alrededor de un m\'odulo compacto y reemplazable.
        \end{itemize}
        \vspace{0.15cm}
        \begin{block}{Decisi\'on de dise\~no}
            \scriptsize La v1 del carrier usa XIAO S3 en formato DIP/socket para facilitar reemplazo y debug; versiones posteriores pueden migrar a castellated/SMD.
        \end{block}
        \column{0.48\textwidth}
        \scriptsize
        \textbf{Asignaci\'on principal de pines}
        \vspace{0.1cm}

        \begin{tabular}{l l}
            \toprule
            \textbf{Funci\'on} & \textbf{GPIO XIAO S3} \\
            \midrule
            ECG OUT & GPIO1 / ADC1\_CH0 \\
            ECG SDN & GPIO2 \\
            Backlight PWM & GPIO3 \\
            SPI DC & GPIO4 \\
            I2C SDA/SCL & GPIO5 / GPIO6 \\
            BTN NEXT & GPIO43 \\
            SPI CS & GPIO44 \\
            SPI SCK/MOSI & GPIO7 / GPIO9 \\
            BTN SELECT & GPIO8 \\
            \bottomrule
        \end{tabular}
        \vspace{0.2cm}
    \end{columns}
\end{frame}

\begin{frame}{Protocolo BLE NimBLE \& Filtro de Calidad}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \footnotesize
        \textbf{Canales de Comunicaci\'on BLE}
        \vspace{0.1cm}
        
        \scriptsize
        \begin{tabular}{l c l}
            \toprule
            \textbf{Caracter\'istica} & \textbf{Flujo} & \textbf{Funci\'on / Cadencia} \\
            \midrule
            \texttt{0xFF01} & Reloj \(\to\) App & IMU continuo (12.5--50\,Hz) \\
            \texttt{0xFF02} & Reloj \(\to\) App & TLV Biometr\'ia agregada (1\,s) \\
            \texttt{0xFF03} & Reloj \(\to\) App & ECG continuous raw (500\,Hz) \\
            \texttt{0xFF04} & App \(\to\) Reloj & Cmd RX (\texttt{0x01}: ON, \texttt{0x00}: OFF) \\
            \bottomrule
        \end{tabular}
        
        \vspace{0.2cm}
        \footnotesize
        El protocolo \textbf{TLV} (Type-Length-Value) permite empaquetar de forma compacta m\'ultiples sensores heterog\'eneos y timestamps en un solo paquete BLE, reduciendo el overhead de radio.
        
        \column{0.52\textwidth}
        \footnotesize
        \textbf{Filtro de Calidad (Quality Gate)}
        \begin{itemize}\itemsep0.1cm
            \item \textbf{Protecci\'on contra falsos positivos}: Como el wearable actual está en protoboard, el contacto sensor-piel puede tener imperfecciones.
            \item \textbf{Byte de Calidad del TLV}: Cada m\'etrica transmitida del firmware posee un byte de calidad. La App solo archiva y eval\'ua alertas si la calidad es \(\ge 60\).
            \item \textbf{Ventana de Estabilidad}: Una alerta cl\'inica requiere que el valor an\'omalo persista por al menos \textbf{60\,s} antes de gatillar una notificaci\'on local, filtrando picos transitorios.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Evoluci\'on del Procesamiento Edge (C3 vs. S3)}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \scriptsize
        \textbf{Restricciones en la ESP32-C3}
        \begin{itemize}\itemsep0.1cm
            \item \textbf{CPU}: Single-core RISC-V a 160\,MHz. Carece de extensiones vectoriales/SIMD por hardware.
            \item \textbf{Memoria}: SRAM muy limitada. Despu\'es de inicializar NimBLE y LVGL, el heap disponible era de tan s\'olo $\sim$99\,KB.
            %\item \textbf{Inferencia}: Demora entre 50 y 200\,ms, forzando un cuello de botella que bloqueaba la GUI y causaba desconexiones en la transmisi\'on BLE.
            \item \textbf{Modelo}: Restringido a un tama\~no $<30$\,KB de Flash y $\leq 10$\,KB de SRAM para no agotar el heap.
        \end{itemize}
        
        \column{0.50\textwidth}
        \scriptsize
        \textbf{Oportunidades con Seeed XIAO S3}
        \begin{itemize}\itemsep0.1cm
            \item \textbf{CPU}: Dual-core Xtensa LX7 a 240\,MHz con FPU e instrucciones vectoriales dedicadas (\textbf{ESP-NN}).
            \item \textbf{Memoria}: 8\,MB Flash y 8\,MB de \textbf{PSRAM} externa octal.
            %\item \textbf{Inferencia}: Reducida a \textbf{5--20\,ms} gracias a la aceleraci\'on SIMD.
            \item \textbf{Concurrencia}: La inferencia corre as\'incronamente en el \textbf{Core 1}, aislando la ejecuci\'on de la GUI y la telemetr\'ia BLE que corren en el \textbf{Core 0}.
            \item \textbf{Modelo}: Permite una arquitectura profunda ($\sim$58\,KB cuantizada) usando un Tensor Arena de 128\,KB alocado en PSRAM.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Arquitectura del Modelo de ML (CNN 1D)}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \scriptsize
        \textbf{Dise\~no de la Red Convolucional}
        \begin{itemize}\itemsep0.08cm
            \item \textbf{Entrada}: Tensor de \((200 \times 6)\) correspondiente a aceler\'ometro y giroscopio (ejes $X, Y, Z$).
            \item \textbf{Capas Convolucionales}: 3 capas Conv1D con filtros progresivos (32, 64, 128) y kernels de tama\~no 5 y 3 para capturar patrones inerciales temporales.
            \item \textbf{Pooling \& Regularizaci\'on}: Global Average Pooling (GAP) para colapsar la dimensi\'on temporal y evitar sobreajuste, m\'as Dropout de 0.3 en la capa densa.
            \item \textbf{Salida}: Softmax de 4 clases:
                \begin{enumerate}\scriptsize\itemsep0.02cm
                    \item \textbf{Reposo} (estado estacionario)
                    \item \textbf{Caminar} (patr\'on c\'iclico de baja frecuencia)
                    \item \textbf{Correr} (patr\'on c\'iclico de alta frecuencia)
                    \item \textbf{Ca\'ida} (evento de impacto y aceleraci\'on extrema)
                \end{enumerate}
        \end{itemize}

\begin{frame}{Pipeline de Datos e Integraci\'on en Firmware}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \scriptsize
        \textbf{Flujo del Procesamiento Local}
        \begin{itemize}\itemsep0.08cm
            \item \textbf{Muestreo}: El IMU BMI160 opera a una frecuencia f\'isica de 100\,Hz.
            \item \textbf{Downsampling}: Un acumulador en la \texttt{har\_task} promedia cada 2 muestras f\'isicas consecutivas, generando una se\~nal filtrada a \textbf{50\,Hz}.
            \item \textbf{Ring Buffer}: Las muestras a 50\,Hz ingresan a un buffer circular est\'atico de 2.4\,KB en SRAM para evitar fragmentar memoria.
            \item \textbf{Normalizaci\'on}: Escalamiento a flotantes dividiendo los valores enteros crudos por 32768.0 (\(\pm 32768\) a \([-1.0, 1.0]\)).
            \item \textbf{Inferencia as\'incrona}: El int\'erprete TFLite Micro ejecuta la inferencia a trav\'es de un blob binario compilado en el firmware (\texttt{har\_model.c}).
            \item \textbf{Alocaci\'on}: Tensor Arena de 128\,KB asignado en \textbf{PSRAM} mediante \texttt{MALLOC\_CAP\_SPIRAM}.
        \end{itemize}
